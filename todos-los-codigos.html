CODIGO ARDUINO PARA EL CONTROL GENERATIVO DE LA IMPRESORA
// === CONTROL GENERATIVO DE 2 EJES (X, Y) ===
// Cada eje tiene 4 arrays y 4 índices independientes
// Cada índice cambia su valor con un ritmo propio, recalculado en cada cambio
// El eje X repite cada movimiento 2 veces (ida y vuelta)

#define X_STEP_PIN 2
#define X_DIR_PIN  5
#define Y_STEP_PIN 3
#define Y_DIR_PIN  6
#define EN_PIN     8  // enable común

// === ARRAYS DE VALORES ===

// EJE X
int time_between_steps_X[] = {100 , 2000, 1000, 500, 200, 100, 1500, 2000, 150, 1500, 120, 100, 150};
int num_steps_X[]          = { 800, 1600, 1800, 1200, 600, 1000};
bool direction_X[]         = {LOW, HIGH, LOW, HIGH};
bool enable_X[]            = {true, true, false, true};

// EJE Y
int time_between_steps_Y[] = {2000, 1000, 500, 200, 100, 1500, 2000, 150, 1500, 120, 100, 150};
int num_steps_Y[]          = {200, 400, 800, 1600, 200, 400, 200, 400, 800, 1600, 800, 1600};
bool direction_Y[]         = {HIGH, LOW, HIGH, LOW, HIGH, HIGH, LOW, HIGH, LOW, HIGH, HIGH, HIGH};
bool enable_Y[]            = {true, true, true, false};

// === ÍNDICES INDEPENDIENTES ===
int idx_time_X = 0, idx_steps_X = 0, idx_dir_X = 0, idx_enable_X = 0;
int idx_time_Y = 0, idx_steps_Y = 0, idx_dir_Y = 0, idx_enable_Y = 0;

// === CONTROL DE MOVIMIENTO ===
long currentStepX = 0;
long currentStepY = 0;
unsigned long lastStepTimeX = 0;
unsigned long lastStepTimeY = 0;

// Nueva variable para repetir movimiento en X
enum { IDLE_X = 0, MOVIENDO_X = 1 };
int estadoX = IDLE_X;
int faseX = 0; // 0 = primera pasada (ida), 1 = segunda pasada (vuelta)
long targetStepsX = 0;
bool dirEfectivaX = LOW;
// === CONTROL DE CAMBIOS DINÁMICOS ===
unsigned long lastChange_time_X = 0;
unsigned long lastChange_steps_X = 0;
unsigned long lastChange_dir_X = 0;
unsigned long lastChange_enable_X = 0;

unsigned long lastChange_time_Y = 0;
unsigned long lastChange_steps_Y = 0;
unsigned long lastChange_dir_Y = 0;
unsigned long lastChange_enable_Y = 0;

// Intervalos iniciales (en milisegundos)
unsigned long interval_time_X = 1000;
unsigned long interval_steps_X = 1500;
unsigned long interval_dir_X = 2000;
unsigned long interval_enable_X = 2500;

unsigned long interval_time_Y = 1200;
unsigned long interval_steps_Y = 1800;
unsigned long interval_dir_Y = 2200;
unsigned long interval_enable_Y = 2600;

void setup() {
  pinMode(X_STEP_PIN, OUTPUT);
  pinMode(X_DIR_PIN, OUTPUT);
  pinMode(Y_STEP_PIN, OUTPUT);
  pinMode(Y_DIR_PIN, OUTPUT);
  pinMode(EN_PIN, OUTPUT);

  Serial.begin(9600);
  randomSeed(analogRead(A0)); // semilla de aleatoriedad

  Serial.println("Iniciando sistema generativo con repetición en X...");

  applyParameters();
}

void loop() {
  unsigned long now = micros();

  // --- EJE X (nueva lógica basada en estado) ---
if (enable_X[idx_enable_X]) {
  unsigned long nowMicros = micros();

  // Si estamos idle, iniciamos un nuevo ciclo (primera pasada)
  if (estadoX == IDLE_X) {
    // Fijar objetivo y dirección efectiva
    targetStepsX = num_steps_X[idx_steps_X];
    dirEfectivaX = direction_X[idx_dir_X]; // dirección original para la primera pasada
    digitalWrite(X_DIR_PIN, dirEfectivaX);
    currentStepX = 0;
    faseX = 0;
    estadoX = MOVIENDO_X;
    // opción: imprimir inicio de ciclo
    Serial.print("X: inicio ciclo -> steps=");
    Serial.println(targetStepsX);
  }

  // Si estamos moviendo, generar pasos según time_between_steps
  if (estadoX == MOVIENDO_X) {
    if (nowMicros - lastStepTimeX >= time_between_steps_X[idx_time_X]) {
      lastStepTimeX = nowMicros;
      if (currentStepX < targetStepsX) {
        digitalWrite(X_STEP_PIN, HIGH);
        delayMicroseconds(2);
        digitalWrite(X_STEP_PIN, LOW);
        currentStepX++;
      } else {
        // completó la pasada actual
        if (faseX == 0) {
          // pasar a segunda pasada: invertir dirección y repetir
          faseX = 1;
          dirEfectivaX = !direction_X[idx_dir_X]; // INVERSA respecto a la original
          digitalWrite(X_DIR_PIN, dirEfectivaX);
          currentStepX = 0;
          Serial.println("X: pasada 1 completa -> iniciando vuelta (invertida)");
        } else {
          // ya hizo ida y vuelta, terminar ciclo
          estadoX = IDLE_X;
          faseX = 0;
          currentStepX = 0;
          // opcional: avanzar índices al terminar ciclo
          Serial.println("X: pasada 2 completa -> ciclo finalizado");
        }
      }
    }
  }

} else {
  // Si enable está OFF, dejamos todo IDLE
  estadoX = IDLE_X;
  currentStepX = 0;
}

  // --- EJE Y ---
  if (enable_Y[idx_enable_Y]) {
    if (now - lastStepTimeY >= time_between_steps_Y[idx_time_Y]) {
      lastStepTimeY = now;
      if (currentStepY < num_steps_Y[idx_steps_Y]) {
        digitalWrite(Y_STEP_PIN, HIGH);
        delayMicroseconds(2);
        digitalWrite(Y_STEP_PIN, LOW);
        currentStepY++;
      } else {
        currentStepY = 0;
      }
    }
  }

  // --- CAMBIOS DE PARÁMETROS ---
  unsigned long t = millis();

  // EJE X
  if (t - lastChange_time_X > interval_time_X)
    changeIndex(idx_time_X, lastChange_time_X, interval_time_X, sizeof(time_between_steps_X) / sizeof(time_between_steps_X[0]));
  if (t - lastChange_steps_X > interval_steps_X)
    changeIndex(idx_steps_X, lastChange_steps_X, interval_steps_X, sizeof(num_steps_X) / sizeof(num_steps_X[0]));
  if (t - lastChange_dir_X > interval_dir_X)
    changeIndex(idx_dir_X, lastChange_dir_X, interval_dir_X, sizeof(direction_X) / sizeof(direction_X[0]));
  if (t - lastChange_enable_X > interval_enable_X)
    changeIndex(idx_enable_X, lastChange_enable_X, interval_enable_X, sizeof(enable_X) / sizeof(enable_X[0]));

  // EJE Y
  if (t - lastChange_time_Y > interval_time_Y)
    changeIndex(idx_time_Y, lastChange_time_Y, interval_time_Y, sizeof(time_between_steps_Y) / sizeof(time_between_steps_Y[0]));
  if (t - lastChange_steps_Y > interval_steps_Y)
    changeIndex(idx_steps_Y, lastChange_steps_Y, interval_steps_Y, sizeof(num_steps_Y) / sizeof(num_steps_Y[0]));
  if (t - lastChange_dir_Y > interval_dir_Y)
    changeIndex(idx_dir_Y, lastChange_dir_Y, interval_dir_Y, sizeof(direction_Y) / sizeof(direction_Y[0]));
  if (t - lastChange_enable_Y > interval_enable_Y)
    changeIndex(idx_enable_Y, lastChange_enable_Y, interval_enable_Y, sizeof(enable_Y) / sizeof(enable_Y[0]));

  applyParameters();
}

// ====================================================
//        FUNCIONES AUXILIARES
// ====================================================

void changeIndex(int &index, unsigned long &lastChange, unsigned long &interval, int cantidadDeDatos) {
  index = (index + 1) % cantidadDeDatos;
  lastChange = millis();
  interval = 500UL * pow(2, random(5)); // 500, 1000, 2000, 4000, 8000 ms
}

void applyParameters() {
  // Si el eje X está en reposo (idle), usamos la dirección del índice.
  // Si está moviendo, mantenemos la dirección efectiva actual.
  if (estadoX == IDLE_X) {
    dirEfectivaX = direction_X[idx_dir_X];
    digitalWrite(X_DIR_PIN, dirEfectivaX);
  } else {
    digitalWrite(X_DIR_PIN, dirEfectivaX);
  }

  // Eje Y: comportamiento normal, independiente
  digitalWrite(Y_DIR_PIN, direction_Y[idx_dir_Y]);

  // Enable global: activo si cualquiera de los dos ejes está habilitado
  bool en = enable_X[idx_enable_X] || enable_Y[idx_enable_Y];
  digitalWrite(EN_PIN, en ? LOW : HIGH);
}

El siguiente es otro código que permite el control de 5 paso a pasos con un RAMPS1.4.
Estoy usandolos para contrloar 3 impresoras en vivo junto a otros 8 canales: 4 motores, 4 solenoides. Este es el dispositivo que uso en vivo. 


#include <MIDI.h>
#include "PololuDriver.h"
#include "Ramps.h"

MIDI_CREATE_DEFAULT_INSTANCE();

Ramps ramps = Ramps();

int microseconds = 25;

unsigned long actual = 0;
// Pines ENABLE
#define EN_X_PIN 38
#define EN_Y_PIN 56
#define EN_Z_PIN 62
#define EN_E_PIN 24
#define EN_Q_PIN 30

// 5 motores: X, Y, Z, E, Q (notas 10 a 14)
bool notePlayed[5] = {false, false, false, false, false};
int motorDir[5] = {1, 1, 1, 1, 1};
int motorDelay[5] = {
  microseconds * 2,
  microseconds * 2,
  microseconds * 16,
  microseconds * 8,
  microseconds * 32
};

int mDm[5] = { 1,  1,  1,  1,  1}; //motor delay multiplicador
int mDmT = 1; // multiplicador total del motor delay

boolean fijo = false;
const int NUM_RELAYS = 8;

int relayNotes[NUM_RELAYS] = {17, 23, 25, 27, 29, 31, 33, 35};
unsigned long ultimoCambio[] = { 0, 0, 0, 0, 0, 0, 0, 0};
bool isON[] = { false, false, false, false, false, false, false, false};

void handleNoteOn(byte channel, byte pitch, byte velocity) {

  for (int i = 0; i < NUM_RELAYS; i++) {
    if (pitch == relayNotes[i]) {
      digitalWrite(pitch, LOW);  // Activar relé
      if (i >= 4 && i < 8) {
        isON[i] = true;
        ultimoCambio[i] = millis();
      }
      break;
    }
  }
  if (pitch == 40) {
    fijo = true;
  }
  if (pitch == 41) {
    mDmT = 1 + (  (mDmT + 1) % 16   );

  }

  mDm[int(random(5))] =    1 + int(random(8));

  if (pitch == 10) digitalWrite(EN_X_PIN, LOW);
  if (pitch == 11) digitalWrite(EN_Y_PIN, LOW);
  if (pitch == 12) digitalWrite(EN_Z_PIN, LOW);
  if (pitch == 13) digitalWrite(EN_E_PIN, LOW);
  if (pitch == 14) digitalWrite(EN_Q_PIN, LOW);  // nuevo motor

  if (pitch >= 10 && pitch <= 14) {
    int index = pitch - 10;
    notePlayed[index] = true;

    // Cambiar dirección individual
    motorDir[index] *= -1;

    ramps.led(true);

    // Setear la dirección del motor correspondiente
    switch (index) {
      case 0:
        ramps.motorX.setDir(motorDir[0]);
        break;
      case 1:
        ramps.motorY.setDir(motorDir[1]);
        break;
      case 2:
        ramps.motorZ.setDir(motorDir[2]);
        break;
      case 3:
        ramps.motorE.setDir(motorDir[3]);
        break;
      case 4:
        ramps.motorQ.setDir(motorDir[4]);
        break;
    }
  }
}

void handleNoteOff(byte channel, byte pitch, byte velocity) {

  for (int i = 0; i < NUM_RELAYS; i++) {
    if (pitch == relayNotes[i] || pitch == 37) {
      digitalWrite(relayNotes[i], HIGH);  // Desactivar relé
      if (i >= 4 && i < 8) {
        isON[i] = false;
        ultimoCambio[i] = millis();
      }
      break;
    }
  }

  if (pitch == 40) {
    fijo = false;
  }

  if (pitch == 41) {
    mDmT = 1 + (  (mDmT + 1) % 16   );
  }

  if (pitch >= 10 && pitch <= 14) {
    int index = pitch - 10;
    notePlayed[index] = false;

    if (!notePlayed[0]) digitalWrite(EN_X_PIN, HIGH);
    if (!notePlayed[1]) digitalWrite(EN_Y_PIN, HIGH);
    if (!notePlayed[2]) digitalWrite(EN_Z_PIN, HIGH);
    if (!notePlayed[3]) digitalWrite(EN_E_PIN, HIGH);
    if (!notePlayed[4]) digitalWrite(EN_Q_PIN, HIGH);

    if (!notePlayed[0] && !notePlayed[1] && !notePlayed[2] && !notePlayed[3] && !notePlayed[4]) {
      ramps.led(false);
    }
  }
}

void setup() {
  MIDI.setHandleNoteOn(handleNoteOn);
  MIDI.setHandleNoteOff(handleNoteOff);

  MIDI.begin();
  Serial.begin(115200);    //descomentar si no se usa mocolufa

  pinMode(EN_X_PIN, OUTPUT);
  pinMode(EN_Y_PIN, OUTPUT);
  pinMode(EN_Z_PIN, OUTPUT);
  pinMode(EN_E_PIN, OUTPUT);
  pinMode(EN_Q_PIN, OUTPUT);

  digitalWrite(EN_X_PIN, HIGH);
  digitalWrite(EN_Y_PIN, HIGH);
  digitalWrite(EN_Z_PIN, HIGH);
  digitalWrite(EN_E_PIN, HIGH);
  digitalWrite(EN_Q_PIN, HIGH);

  for (int i = 0; i < NUM_RELAYS; i++) {
    pinMode(relayNotes[i], OUTPUT);
    digitalWrite(relayNotes[i], HIGH);  // Desactivado inicialmente
  }
}

void loop() {
  // cambiarMicroSeconds();
  stepsOnOff();

  MIDI.read();

  if (notePlayed[0] || notePlayed[1] || notePlayed[2] || notePlayed[3] || notePlayed[4]) {
    ramps.led(true);
  } else {
    ramps.led(false);
  }



  actual = millis();
  for (int i = 0; i < 8; i++) {
    if (isON[i] = true)
    {
      if ( (actual - ultimoCambio[i] ) > 100) {
        ultimoCambio[i] = actual;
        isON[i] = false;
        digitalWrite(relayNotes[i], HIGH);
      }
    }
  }

}

}

void stepsOnOff() {
  if (fijo) {
    pasosFijos();
  }
  else {
    pasosCambiantes();
  }
}

void ledrefe() {
  if (millis() % 100 < 50)
  {
    ramps.led(true);
  }
  else
  {
    ramps.led(false);
  }
}


void pasosCambiantes() {
  if (notePlayed[0]) { // nota 10 - motor X
    ramps.motorX.stepOn();
    delayMicroseconds(motorDelay[0]*mDm[0]);
    ramps.motorX.stepOff();
  }
  if (notePlayed[1]) { // nota 11 - motor Y
    ramps.motorY.stepOn();
    delayMicroseconds(motorDelay[1]*mDm[1]);
    ramps.motorY.stepOff();
  }
  if (notePlayed[2]) { // nota 12 - motor Z
    ramps.motorZ.stepOn();
    delayMicroseconds(motorDelay[2]*mDm[2]);
    ramps.motorZ.stepOff();
  }
  if (notePlayed[3]) { // nota 13 - motor E
    ramps.motorE.stepOn();
    delayMicroseconds(motorDelay[3]*mDm[3]);
    ramps.motorE.stepOff();
  }
  if (notePlayed[4]) { // nota 14 - motor Q
    ramps.motorQ.stepOn();
    delayMicroseconds(motorDelay[4]*mDm[4]);
    ramps.motorQ.stepOff();
  }
}



void pasosFijos() {

  int MDMTfinal =  map(mDmT, 0,16,16,0); 
  if (notePlayed[0]) { // nota 10 - motor X
    ramps.motorX.stepOn();
    delayMicroseconds(motorDelay[0]*MDMTfinal);
    ramps.motorX.stepOff();
  }
  if (notePlayed[1]) { // nota 11 - motor Y
    ramps.motorY.stepOn();
    delayMicroseconds(motorDelay[1]*MDMTfinal);
    ramps.motorY.stepOff();
  }
  if (notePlayed[2]) { // nota 12 - motor Z
    ramps.motorZ.stepOn();
    delayMicroseconds(motorDelay[2]*MDMTfinal);
    ramps.motorZ.stepOff();
  }
  if (notePlayed[3]) { // nota 13 - motor E
    ramps.motorE.stepOn();
    delayMicroseconds(motorDelay[3]*MDMTfinal);
    ramps.motorE.stepOff();
  }
  if (notePlayed[4]) { // nota 14 - motor Q
    ramps.motorQ.stepOn();
    delayMicroseconds(motorDelay[4]*MDMTfinal);
    ramps.motorQ.stepOff();
  }
}
